cmake_minimum_required(VERSION 3.18)
project(hoge_f2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TFHEPP_MARCH "x86-64-v3" CACHE STRING "CPU architecture for both TFHEpp and its caller")
set(USE_AVX512 OFF CACHE BOOL "F2 host uses AVX2")
set(AWS_SDK_DIR "$ENV{SDK_DIR}" CACHE PATH "AWS FPGA SDK directory")
find_path(AWS_INCLUDE fpga_mgmt.h HINTS "${AWS_SDK_DIR}/userspace/include" REQUIRED)
find_library(AWS_MGMT fpga_mgmt HINTS "${AWS_SDK_DIR}/userspace/lib" "${AWS_SDK_DIR}/userspace/lib/so" /usr/local/lib64 /usr/local/lib REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenMP)
add_subdirectory(../thirdparties/TFHEpp "${CMAKE_CURRENT_BINARY_DIR}/TFHEpp")
add_executable(nand_f2 ../xcltest/HomGate/nand.cpp software/fpga_backend.cpp)
target_compile_definitions(nand_f2 PRIVATE HOGE_AWS_F2=1 ${TFHEpp_DEFINITIONS})
target_compile_options(nand_f2 PRIVATE "-march=${TFHEPP_MARCH}")
target_include_directories(nand_f2 PRIVATE "${AWS_INCLUDE}")
target_link_libraries(nand_f2 PRIVATE tfhe++ "${AWS_MGMT}" Threads::Threads)
if(OpenMP_CXX_FOUND)
    target_link_libraries(nand_f2 PRIVATE OpenMP::OpenMP_CXX)
endif()

# Export reference vectors using an isolated, checked adaptation of the NAND test.
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(VECTOR_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/nand_vector_source.cpp")
add_custom_command(OUTPUT "${VECTOR_SOURCE}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/sim/make_vector_source.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/../xcltest/HomGate/nand.cpp" "${VECTOR_SOURCE}"
  DEPENDS sim/make_vector_source.py ../xcltest/HomGate/nand.cpp VERBATIM)
add_executable(nand_vectors "${VECTOR_SOURCE}" sim/vector_export.cpp)
target_compile_options(nand_vectors PRIVATE "-march=${TFHEPP_MARCH}")
target_compile_definitions(nand_vectors PRIVATE ${TFHEpp_DEFINITIONS})
target_link_libraries(nand_vectors PRIVATE tfhe++)
if(OpenMP_CXX_FOUND)
  target_link_libraries(nand_vectors PRIVATE OpenMP::OpenMP_CXX)
endif()
